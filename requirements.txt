# =============================================================================
# requirements.txt
#
# Python dependencies for the Spherical Tokamak shielding neutronics study
# (vvij_ptnr_2023.ipynb), reproducing the fusion-energy/neutronics-workshop
# package set, updated to match the notebook's current paramak (>=0.9.8) API.
#
# IMPORTANT — OpenMC / MOAB source:
# The plain PyPI `openmc` wheel does NOT ship with DAGMC support compiled in.
# Instead, this project uses Jonathan Shimwell's (fusion-energy) community
# wheel index, which hosts prebuilt OpenMC wheels WITH DAGMC/MOAB/Embree/
# Double-Down support already built in, plus a matching pymoab wheel. This is
# the same trick the neutronics-workshop ecosystem itself uses to avoid
# compiling DAGMC/MOAB from source on every build:
#
#   pip install --extra-index-url https://shimwell.github.io/wheels openmc
#
# =============================================================================

--extra-index-url https://shimwell.github.io/wheels

# --- Core neutronics / CAD stack (DAGMC-enabled wheels) --------------------
openmc                              # DAGMC-enabled build, from shimwell's wheel index (see above)
moab                                 # pymoab bindings, from the same index — needed by cad_to_dagmc
cadquery                             # paramak's CAD backend (official PyPI wheels since 2.4)
paramak>=0.9.8                       # current function-based API (spherical_tokamak, revolved_shape,
                                      # center_column_shield_cylinder, blanket_from_plasma, plasma_simplified, ...)
                                      # NOTE: do NOT pin this to 0.8.x — that's the old class-based API
                                      # (RotateStraightShape / Reactor / .show()) this notebook no longer uses.
cad_to_dagmc                         # CadQuery Assembly/Workplane -> DAGMC h5m (replaces stl_to_h5m + pydagmc)
openmc_data_downloader                # on-the-fly nuclear data fetching helper
openmc_tally_unit_converter           # tally unit conversion (eV -> MeV -> W/g etc.)
openmc_plasma_source                  # optional: realistic plasma neutron sources
openmc_source_plotter                 # plots source particle locations/directions/energies
openmc_depletion_plotter               # plots isotopes present during depletion steps
openmc_regular_mesh_plotter             # publication-quality regular mesh tally plots
neutronics_material_maker[density]      # material recipes / density lookups, exports to OpenMC etc.

# --- gmsh (used internally by cad_to_dagmc's meshing backend, and directly in the notebook) --
gmsh

# --- Scientific / numerical stack -------------------------------------------
numpy
scipy
sympy
mpmath
pandas

# --- Plotting / imaging -------------------------------------------------
matplotlib
Pillow
